Module 6.8 — Graphs

A graph is a set of nodes (vertices) connected by edges, representing relationships that aren't strictly hierarchical (unlike trees). Graphs model an enormous range of real problems — networks, dependencies, maps — and BFS/DFS are the foundation nearly every graph algorithm builds on.

1. Graph Representations

Adjacency List (most common — efficient for sparse graphs)

int n = 5;   // number of nodes
vector<vector<int>> adj(n);   // adj[u] = list of nodes directly connected to u

// Adding an undirected edge between u and v
void addEdge(vector<vector<int>> &adj, int u, int v) {
    adj[u].push_back(v);
    adj[v].push_back(u);   // omit this line for a DIRECTED edge (u -> v only)
}

addEdge(adj, 0, 1);
addEdge(adj, 0, 2);
addEdge(adj, 1, 3);
// adj[0] = {1, 2}, adj[1] = {0, 3}, adj[2] = {0}, adj[3] = {1}

Adjacency Matrix (simpler, but wasteful for sparse graphs)

vector<vector<int>> matrix(n, vector<int>(n, 0));
matrix[0][1] = 1;   // edge from 0 to 1
matrix[1][0] = 1;      // undirected — mirror it

// matrix[u][v] == 1 means an edge exists between u and v — O(1) EDGE CHECK,
// but O(V²) SPACE regardless of how many edges actually exist

Adjacency List vs Matrix — When to Use Which

AspectAdjacency listAdjacency matrix
SpaceO(V + E)O(V²), regardless of actual edge count
Check if edge (u,v) existsO(degree of u)O(1)
Iterate all neighbors of a nodeO(degree of u) — efficientO(V) — must scan the entire row
Best forSparse graphs (most real-world graphs)Dense graphs, or when O(1) edge lookup matters most

Weighted Graphs

vector<vector<pair<int,int>>> adj(n);   // adj[u] = list of (neighbor, weight) pairs — using pair, Module 4.2

void addWeightedEdge(vector<vector<pair<int,int>>> &adj, int u, int v, int w) {
    adj[u].push_back({v, w});
    adj[v].push_back({u, w});
}

2. Depth-First Search (DFS)

void dfs(int node, vector<vector<int>> &adj, vector<bool> &visited) {
    visited[node] = true;
    cout << node << " ";

    for (int neighbor : adj[node]) {
        if (!visited[neighbor]) {
            dfs(neighbor, adj, visited);
        }
    }
}
// O(V + E) time — every vertex and every edge is examined exactly once
// O(V) space — the recursive call stack, worst case as deep as the number of vertices

Iterative DFS (using an explicit stack — same pattern as tree DFS in Module 6.6)

void dfsIterative(int start, vector<vector<int>> &adj) {
    vector<bool> visited(adj.size(), false);
    stack<int> st;
    st.push(start);

    while (!st.empty()) {
        int node = st.top();
        st.pop();

        if (visited[node]) continue;   // may be pushed multiple times before being visited — skip duplicates
        visited[node] = true;
        cout << node << " ";

        for (int neighbor : adj[node]) {
            if (!visited[neighbor]) st.push(neighbor);
        }
    }
}

3. Breadth-First Search (BFS)

void bfs(int start, vector<vector<int>> &adj) {
    vector<bool> visited(adj.size(), false);
    queue<int> q;

    visited[start] = true;
    q.push(start);

    while (!q.empty()) {
        int node = q.front();
        q.pop();
        cout << node << " ";

        for (int neighbor : adj[node]) {
            if (!visited[neighbor]) {
                visited[neighbor] = true;   // mark visited AT PUSH TIME, not pop time — critical detail below
                q.push(neighbor);
            }
        }
    }
}
// O(V + E) time, O(V) space

Critical detail — mark visited when PUSHING, not when POPPING: if you wait until popping to mark a node visited, the same node can be pushed onto the queue multiple times (once for each edge leading to it) before it's ever processed — wasting time and, in some problems, causing incorrect results. Marking at push time guarantees each node enters the queue exactly once.

BFS Guarantees Shortest Path in UNWEIGHTED Graphs

Because BFS explores level by level, the first time it reaches any node is guaranteed to be via the shortest possible path (fewest edges) from the start — this is why BFS, not DFS, is used for shortest-path problems on unweighted graphs. 

int shortestPath(int start, int end, vector<vector<int>> &adj) {
    vector<int> dist(adj.size(), -1);
    queue<int> q;

    dist[start] = 0;
    q.push(start);

    while (!q.empty()) {
        int node = q.front();
        q.pop();

        if (node == end) return dist[node];   // found it — this IS the shortest distance, guaranteed

        for (int neighbor : adj[node]) {
            if (dist[neighbor] == -1) {   // not yet visited
                dist[neighbor] = dist[node] + 1;
                q.push(neighbor);
            }
        }
    }
    return -1;   // unreachable
}

5. Connected Components

int countComponents(int n, vector<vector<int>> &adj) {
    vector<bool> visited(n, false);
    int components = 0;

    for (int i = 0; i < n; i++) {
        if (!visited[i]) {
            dfs(i, adj, visited);   // explores the ENTIRE component containing i
            components++;
        }
    }
    return components;
}
// O(V + E) — each DFS call, summed across all starting points, still only visits each node/edge once total

6. Cycle Detection

In an Undirected Graph

bool hasCycleUndirected(int node, int parent, vector<vector<int>> &adj, vector<bool> &visited) {
    visited[node] = true;

    for (int neighbor : adj[node]) {
        if (!visited[neighbor]) {
            if (hasCycleUndirected(neighbor, node, adj, visited)) return true;
        } else if (neighbor != parent) {
            // visited AND not the node we just came from — a back edge exists — CYCLE
            return true;
        }
    }
    return false;
}

In a Directed Graph (needs a separate "in current recursion path" tracker)

bool hasCycleDirected(int node, vector<vector<int>> &adj, vector<bool> &visited, vector<bool> &inStack) {
    visited[node] = true;
    inStack[node] = true;

    for (int neighbor : adj[node]) {
        if (!visited[neighbor]) {
            if (hasCycleDirected(neighbor, adj, visited, inStack)) return true;
        } else if (inStack[neighbor]) {
            // neighbor is CURRENTLY on the recursion stack — a back edge WITHIN the current path — CYCLE
            return true;
        }
    }

    inStack[node] = false;   // backtrack — this node is no longer on the current path
    return false;
}
// A separate inStack tracker is REQUIRED for directed graphs, because a node being "visited"
// overall doesn't mean it's part of a cycle — it might have been fully explored via a
// DIFFERENT path earlier, which is perfectly fine and not a cycle

7. Dijkstra's Shortest Path Algorithm (Weighted Graphs)

Extends BFS's shortest-path idea to weighted graphs (with non-negative weights), using a min-heap (Module 6.7) to always process the currently-closest unvisited node next — a greedy approach. 

vector<int> dijkstra(int start, int n, vector<vector<pair<int,int>>> &adj) {
    vector<int> dist(n, INT_MAX);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>>> pq;  // min-heap of (distance, node)

    dist[start] = 0;
    pq.push({0, start});

    while (!pq.empty()) {
        auto [d, node] = pq.top();   // structured binding — Module 5.6
        pq.pop();

        if (d > dist[node]) continue;   // a shorter path to `node` was already found — this entry is stale, skip it

        for (auto &[neighbor, weight] : adj[node]) {
            int newDist = dist[node] + weight;
            if (newDist < dist[neighbor]) {
                dist[neighbor] = newDist;
                pq.push({newDist, neighbor});
            }
        }
    }
    return dist;
}
// O((V + E) log V) time — each edge may trigger a heap push, each push/pop is O(log V)
// This directly reuses the priority_queue mechanics from Module 4.3 and the heap
// internals from Module 4.8/6.7 — a natural capstone application of prior modules

Why this only works with non-negative weights: the greedy assumption — that once a node is popped with its minimum distance, that distance can never be improved — breaks down if an edge could have negative weight, since a longer path taken later could still end up cheaper overall. (Graphs with negative weights require Bellman-Ford instead, a stretch topic beyond this module's core scope.)
